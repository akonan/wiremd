🎨 wiremd - Simple Browser Examples

wiremd can now run entirely in your browser! Below are copy-ready examples using UMD so they also work when opened directly via file://.

Example 1: File-safe UMD Import

The simplest way to run locally without a dev server:

<!DOCTYPE html>
<html>
<head>
  <title>wiremd Example</title>
</head>
<body>
  <iframe id="preview" style="width:100%;height:420px;border:1px solid #ddd;"></iframe>

  <script src="./dist/wiremd.umd.js"></script>
  <script>
    const { parse, renderToHTML } = window.wiremd;

    const markdown = `
# Login Form

Username
[_____________________]

Password  
[_____________________]

[Sign In]{.primary} [Forgot Password?]
`;

    const ast = parse(markdown);
    const html = renderToHTML(ast, { style: 'clean', inlineStyles: true });
    document.getElementById('preview').srcdoc = html;
  </script>
</body>
</html>
Steps:
  1. npm run build:browser
  2. Save the code above as example.html
  3. Open in your browser

Example 2: Live Rendering with Real-time Updates

Create a markdown editor with live preview:

<!DOCTYPE html>
<html>
<head>
  <title>wiremd Editor</title>
  <style>
    body { font-family: sans-serif; margin: 0; }
    .container { display: grid; grid-template-columns: 1fr 1fr; height: 100vh; }
    textarea { padding: 20px; font-family: monospace; resize: none; }
    #preview { width: 100%; height: 100%; border: 0; background: white; }
  </style>
</head>
<body>
  <div class="container">
    <textarea id="markdown" placeholder="Enter wiremd markdown..."></textarea>
    <iframe id="preview" title="wiremd preview"></iframe>
  </div>

  <script src="./dist/wiremd.umd.js"></script>
  <script>
    const { parse, renderToHTML } = window.wiremd;

    const textArea = document.getElementById('markdown');
    const preview = document.getElementById('preview');

    function render() {
      try {
        const ast = parse(textArea.value);
        const html = renderToHTML(ast, { inlineStyles: true });
        preview.srcdoc = html;
      } catch (e) {
        preview.srcdoc = `<p style="font-family:sans-serif;color:red;padding:12px;">Error: ${e.message}</p>`;
      }
    }

    textArea.addEventListener('input', render);
    textArea.value = `# Hello

## Welcome to wiremd

Create wireframes with Markdown!

[Get Started]{.primary}`;
    render();
  </script>
</body>
</html>

Example 3: React/JSX Export

Generate React components from wiremd markdown:

<!DOCTYPE html>
<html>
<head>
  <title>wiremd React Export</title>
</head>
<body>
  <textarea id="markdown" style="width:400px; height:300px;"></textarea>
  <pre id="output" style="width:400px; height:300px; overflow:auto; background:#f0f0f0;"></pre>

  <script src="./dist/wiremd.umd.js"></script>
  <script>
    const { parse, renderToReact } = window.wiremd;

    const textArea = document.getElementById('markdown');
    const output = document.getElementById('output');

    textArea.value = `## User Profile

Name: [John Doe]
Email: john@example.com

[Edit Profile]{.primary} [Sign Out]`;

    function generate() {
      try {
        const ast = parse(textArea.value);
        const jsx = renderToReact(ast);
        output.textContent = jsx;
      } catch (e) {
        output.textContent = `Error: ${e.message}`;
      }
    }

    textArea.addEventListener('input', generate);
    generate();
  </script>
</body>
</html>

Example 4: Different Styles

Try all the available wireframe styles:

<!DOCTYPE html>
<html>
<head>
  <title>wiremd Styles</title>
  <style>
    body { font-family: sans-serif; padding: 20px; }
    .gallery { display: grid; grid-template-columns: repeat(auto-fit, minmax(300px, 1fr)); gap: 20px; }
    .item { border: 1px solid #ddd; padding: 10px; border-radius: 4px; }
    h3 { margin-top: 0; }
  </style>
</head>
<body>
  <h1>wiremd Styles Gallery</h1>
  <div class="gallery" id="gallery"></div>

  <script src="./dist/wiremd.umd.js"></script>
  <script>
    const { parse, renderToHTML } = window.wiremd;

    const markdown = `
## Contact Form

Email
[_____________________]

[Subscribe]{.primary}
`;

    const styles = ['sketch', 'clean', 'wireframe', 'material', 'tailwind'];
    const gallery = document.getElementById('gallery');

    styles.forEach(style => {
      const ast = parse(markdown);
      const html = renderToHTML(ast, { style, inlineStyles: true });
      
      const div = document.createElement('div');
      div.className = 'item';
      div.innerHTML = `<h3>${style}</h3><iframe style="width:100%;height:260px;border:0;" title="${style} preview"></iframe>`;
      div.querySelector('iframe').srcdoc = html;
      gallery.appendChild(div);
    });
  </script>
</body>
</html>

npm run build

# Build everything
npm run build && npm run build:browser
